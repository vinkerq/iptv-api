# 禁用requests冗余警告，减少IO耗时
requests.packages.urllib3.disable_warnings()
# 全局Session复用连接池，大幅提升请求速度
SESSION = requests.Session()
SESSION.headers = {"User-Agent": "Mozilla/5.0"}  # 精简请求头
SESSION.verify = False  # 关闭SSL验证（提速）
SESSION.allow_redirects = False  # 禁用重定向（提速）

def parse_ip_port(line: str):
    """解析ip.txt每行的IP:端口，提取网段前缀和端口"""
    line = line.strip()
    if not line:
        return None, None
    try:
        ip_part, port = line.split(":", 1)
        port = int(port)
        ip_segs = ip_part.split(".")
        if len(ip_segs) != 4:
            return None, None
        net_prefix = ".".join(ip_segs[:3])  # 取IP前三段（如125.92.140.1 → 125.92.140）
        return net_prefix, port
    except:
        return None, None

def extract_ip_port_from_url(url: str):
    """从完整URL中提取IP:端口（核心修改函数）"""
    parsed = urlparse(url)
    # netloc属性直接返回 "IP:端口" 格式，无需手动切割
    return parsed.netloc

def scan_single_ip(net_prefix: str, host_num: int, port: int):
    """扫描单个IP，极简逻辑提速"""
    full_ip = f"{net_prefix}.{host_num}"
    full_url = f"http://{full_ip}:{port}{RTP_SUFFIX}"
    try:
        # stream=True：不接收完整响应体，只校验状态码，大幅提速
        resp = SESSION.get(full_url, timeout=TIMEOUT, stream=True)
        if 200 <= resp.status_code < 300:
            # 提取IP:端口并返回，而非完整URL
            return extract_ip_port_from_url(full_url)
        return None
    except:
        return None

def run_ip_scan():
    """执行IP扫描主流程"""
    print("="*60)
    print("开始执行IP扫描任务...")
    print("="*60)
    
    # 1. 读取并解析ip.txt
    if not os.path.exists(IP_CONFIG_FILE):
        print(f"❌ 错误：未找到文件 {IP_CONFIG_FILE}")
        return False

    net_port_list = []
    with open(IP_CONFIG_FILE, "r", encoding="utf-8") as f:
        for line in f:
            net_prefix, port = parse_ip_port(line)
            if net_prefix and port:
                net_port_list.append((net_prefix, port))

    if not net_port_list:
        print("❌ 错误：ip.txt中无有效【IP:端口】格式内容")
        return False

    # 2. 生成所有扫描任务
    tasks = []
    for net_prefix, port in net_port_list:
        print(f"📌 待扫描网段：{net_prefix}.1-300:{port}")
        for host_num in range(SCAN_RANGE[0], SCAN_RANGE[1]):
            tasks.append((net_prefix, host_num, port))

    total = len(tasks)
    if total == 0:
        print("❌ 无扫描任务")
        return False
    
    # 3. 启动40线程池扫描
    print(f"\n🚀 开始扫描 | 总任务：{total} | 并发线程：{MAX_WORKERS} | 超时：{TIMEOUT}秒")
    print(f"✅ 仅打印有效IP:端口，精简日志提速\n")
    start_time = time()

    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        # 提交所有任务到线程池
        future_to_task = {executor.submit(scan_single_ip, *t): t for t in tasks}
        # 实时获取完成的任务结果
        for future in as_completed(future_to_task):
            result = future.result()
            if result:
                VALID_IP_PORTS.add(result)
                print(f"✅ 有效IP:端口：{result}")  # 输出仅保留IP:端口

    # 4. 保存结果
    if VALID_IP_PORTS:
        sorted_ip_ports = sorted(VALID_IP_PORTS)
        with open(VALID_IP_FILE, "w", encoding="utf-8") as f:
            f.write("\n".join(sorted_ip_ports))  # 保存仅保留IP:端口
        print(f"\n💾 扫描完成！有效IP:端口已保存至：{VALID_IP_FILE}")
        print(f"📊 统计：总任务{total} | 有效{len(sorted_ip_ports)} | 耗时{time()-start_time:.1f}秒")
        print(f"\n📋 有效IP:端口列表：")
        for ip_port in sorted_ip_ports:
            print(f"✅ {ip_port}")  # 列表展示仅保留IP:端口
    else:
        print(f"\n❌ 扫描完成！无有效IP:端口 | 总耗时{time()-start_time:.1f}秒")
        # 清空旧文件，避免残留数据
        if os.path.exists(VALID_IP_FILE):
            open(VALID_IP_FILE, "w").close()

    # 关闭全局Session
    SESSION.close()
    return True

def get_program_category(program_name):
    """根据节目名称判断所属分类（取消其他频道，全部归入广东频道）"""
    program_name = program_name.lower()  # 转小写，避免大小写问题
    if any(keyword in program_name for keyword in ['cctv', '央视']):
        return "央视频道"
    elif any(keyword in program_name for keyword in ['香港', '澳门', '凤凰', '本港台', '翡翠台']):
        return "港澳频道"
    elif any(keyword in program_name for keyword in ['卫视', '浙江', '江苏', '湖南', '东方', '北京', '安徽']):
        return "卫视频道"
    else:
        # 所有不匹配其他分类的节目都归入广东频道
        return "广东频道"

def get_cctv_sort_key(program_line):
    """提取CCTV节目行的排序键"""
    # 提取节目名称部分
    program_name = program_line.split(',')[0]
    # 匹配CCTV后面的数字（支持CCTV1、CCTV-1、央视1套等格式）
    match = re.search(r'(cctv|央视).*?(\d+)', program_name.lower())
    if match:
        return (int(match.group(2)), program_name)
    # 处理CCTV5+这类特殊频道
    elif '+' in program_name.lower() and 'cctv' in program_name.lower():
        return (int(re.search(r'\d+', program_name).group()) + 0.5, program_name)
    else:
        # 非数字的CCTV频道排在最后
        return (99, program_name)

def get_guangdong_sort_key(program_line):
    """广东频道排序键：以"广"开头的排前面，然后按名称拼音排序"""
    program_name = program_line.split(',')[0]
    # 优先级：以"广"开头的为0（排前面），其他为1（排后面）
    priority = 0 if program_name.startswith('广') else 1
    # 按名称进行自然排序
    return (priority, program_name)

def read_ip_ports(file_path):
    """读取IP端口文件，返回去重后的IP端口列表"""
    ip_ports = set()
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            for line in f:
                line = line.strip()
                if line and ':' in line:
                    ip_part, port_part = line.split(':', 1)
                    if port_part.isdigit():
                        ip_ports.add(line)
    except FileNotFoundError:
        print(f"错误：未找到IP端口文件 {file_path}")
        return []
    except Exception as e:
        print(f"读取IP端口文件时出错：{e}")
        return []
    return list(ip_ports)

def read_program_templates(file_path):
    """读取节目模板文件，返回节目名称和地址模板的字典（节目名作为key）"""
    program_templates = defaultdict(list)  # 键：节目名，值：该节目对应的所有模板地址
    try:
        with open(file_path, 'r', encoding='utf-8') as f:
            for line_num, line in enumerate(f, 1):
                line = line.strip()
                if not line:
                    continue
                if ',' not in line:
                    print(f"警告：第{line_num}行格式错误（无逗号分隔），已跳过：{line}")
                    continue
                name, url_template = line.split(',', 1)
                name = name.strip()
                url_template = url_template.strip()
                if not name or not url_template:
                    print(f"警告：第{line_num}行名称或地址为空，已跳过")
                    continue
                program_templates[name].append(url_template)
    except FileNotFoundError:
        print(f"错误：未找到节目模板文件 {file_path}")
        return {}
    except Exception as e:
        print(f"读取节目模板文件时出错：{e}")
        return {}
    return program_templates

def generate_categorized_list(ip_ports, program_templates, output_path):
    """按分类生成节目列表并写入文件"""
    if not ip_ports or not program_templates:
        print("IP端口列表或节目模板列表为空，无法生成文件")
        return
    
    # 1. 按分类整理节目（移除了其他频道）

        "央视频道": [],
        "卫视频道": [],
        "广东频道": [],
        "港澳频道": []
    }
    
    # 2. 遍历每个节目名，生成所有IP端口的地址并归类
    for program_name, templates in program_templates.items():
        category = get_program_category(program_name)
        # 为该节目生成所有IP端口的地址
        program_lines = []
        for template in templates:
            for ip_port in ip_ports:
                new_url = template.replace("ip地址端口", ip_port)
                program_lines.append(f"{program_name},{new_url}")

        categorized_programs[category].extend(program_lines)
    
    # 3. 对央视频道进行特殊排序（按CCTV1-17顺序）
    if categorized_programs["央视频道"]:
        categorized_programs["央视频道"].sort(key=get_cctv_sort_key)

    if categorized_programs["广东频道"]:
        categorized_programs["广东频道"].sort(key=get_guangdong_sort_key)
    
    # 5. 写入文件
    try:
        with open(output_path, 'w', encoding='utf-8') as f:
            total_lines = 0
            # 按指定顺序写入分类（移除了其他频道）
            for category_key in ["央视频道", "卫视频道", "广东频道", "港澳频道"]:
                programs = categorized_programs[category_key]
                if not programs:
                    continue
                # 写入自定义的分类标题
                category_title = CATEGORY_MAPPING[category_key]
                f.write(f"{category_title}\n")
                # 写入该分类下的所有节目
                for line in programs:
                    f.write(f"{line}\n")
                f.write("\n")  # 分类间空行分隔
                total_lines += len(programs)
        
        print(f"\n✅ 成功生成分类节目列表文件：{output_path}")
        print(f"📊 统计信息：")
        print(f"   - 有效IP端口数量：{len(ip_ports)}")
        print(f"   - 有效节目名称数量：{len(program_templates)}")
        print(f"   - 生成的节目记录总数：{total_lines}")
        # 打印各分类数量
        for category_key in ["央视频道", "卫视频道", "广东频道", "港澳频道"]:
            count = len(categorized_programs[category_key])
            if count > 0:
                print(f"   - {CATEGORY_MAPPING[category_key]}：{count} 条")
    except Exception as e:
        print(f"❌ 写入文件时出错：{e}")
